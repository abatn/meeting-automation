DATABASE CONSTRAINTS & INDICES
================================================================================

meetings table:
  - PK: id
  - FK: client_id → clients.id (CASCADE)
  - FK: creator_id → users.id
  - FK: room_id → meeting_rooms.id
  - CHECK: end_time IS NULL OR end_time > start_time
  - INDEX: (created_at DESC)

participants table:
  - PK: id
  - FK: meeting_id → meetings.id (CASCADE)
  - FK: user_id → users.id
  - UNIQUE: (meeting_id, email)

recordings table:
  - PK: id
  - FK: client_id → clients.id (CASCADE)
  - FK: meeting_id → meetings.id (CASCADE)
  - INDEX: (meeting_id, status)

transcriptions table:
  - PK: id
  - FK: client_id → clients.id (CASCADE)
  - FK: meeting_id → meetings.id (CASCADE)
  - FK: recording_id → recordings.id (CASCADE)
  - Idempotency check: (recording_id, client_id)

actions table:
  - PK: id
  - FK: meeting_id → meetings.id (CASCADE)
  - FK: pv_id → pv.id
  - INDEX: (meeting_id, status)

n8n_meetings table (audit trail):
  - PK: id
  - FK: meeting_id → meetings.id (CASCADE)
